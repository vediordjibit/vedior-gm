// functions/src/index.ts
// Firebase Cloud Function — proxy Anthropic API
// Fix CORS : autorise vediorgm.com + vediorgm.web.app + vediorgm.firebaseapp.com

import * as functions from 'firebase-functions';
import * as admin from 'firebase-admin';

admin.initializeApp();

const ALLOWED_ORIGINS = [
  'https://vediorgm.com',
  'https://www.vediorgm.com',
  'https://vediorgm.web.app',
  'https://vediorgm.firebaseapp.com',
  'http://localhost:3000',
  'http://localhost:3001',
];

// ── Helper CORS headers ───────────────────────────────────────────────────────
function setCorsHeaders(req: functions.https.Request, res: functions.Response) {
  const origin = req.headers.origin as string;
  const allowed = ALLOWED_ORIGINS.includes(origin) ? origin : ALLOWED_ORIGINS[0];

  res.set('Access-Control-Allow-Origin', allowed);
  res.set('Access-Control-Allow-Methods', 'POST, OPTIONS');
  res.set('Access-Control-Allow-Headers', 'Content-Type, Authorization');
  res.set('Access-Control-Max-Age', '3600');
}

// ── Anthropic Proxy ───────────────────────────────────────────────────────────
export const anthropicProxy = functions
  .region('us-central1')
  .runWith({ timeoutSeconds: 60, memory: '256MB' })
  .https.onRequest(async (req, res) => {

    setCorsHeaders(req, res);

    // Preflight OPTIONS
    if (req.method === 'OPTIONS') {
      res.status(204).send('');
      return;
    }

    if (req.method !== 'POST') {
      res.status(405).json({ error: 'Method not allowed' });
      return;
    }

    try {
      const apiKey = process.env.ANTHROPIC_API_KEY
        || functions.config().anthropic?.key
        || functions.config().anthropic?.api_key;

      if (!apiKey) {
        res.status(500).json({ error: 'ANTHROPIC_API_KEY not configured' });
        return;
      }

      const body = req.body;

      const response = await fetch('https://api.anthropic.com/v1/messages', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'x-api-key': apiKey,
          'anthropic-version': '2023-06-01',
        },
        body: JSON.stringify(body),
      });

      const data = await response.json();

      if (!response.ok) {
        res.status(response.status).json(data);
        return;
      }

      res.status(200).json(data);

    } catch (error: any) {
      console.error('Proxy error:', error);
      res.status(500).json({ error: error.message || 'Internal server error' });
    }
  });


// ── Groq Proxy (optionnel — même pattern) ────────────────────────────────────
export const groqProxy = functions
  .region('us-central1')
  .runWith({ timeoutSeconds: 60, memory: '256MB' })
  .https.onRequest(async (req, res) => {

    setCorsHeaders(req, res);

    if (req.method === 'OPTIONS') {
      res.status(204).send('');
      return;
    }

    if (req.method !== 'POST') {
      res.status(405).json({ error: 'Method not allowed' });
      return;
    }

    try {
      const apiKey = process.env.GROQ_API_KEY
        || functions.config().groq?.key
        || functions.config().groq?.api_key;

      if (!apiKey) {
        res.status(500).json({ error: 'GROQ_API_KEY not configured' });
        return;
      }

      const body = req.body;

      const response = await fetch('https://api.groq.com/openai/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${apiKey}`,
        },
        body: JSON.stringify({
          model: body.model || 'llama-3.3-70b-versatile',
          max_tokens: body.max_tokens || 1500,
          temperature: body.temperature || 0.2,
          messages: body.messages,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        res.status(response.status).json(data);
        return;
      }

      res.status(200).json(data);

    } catch (error: any) {
      console.error('Groq proxy error:', error);
      res.status(500).json({ error: error.message || 'Internal server error' });
    }
  });
